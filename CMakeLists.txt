cmake_minimum_required(VERSION 3.21)
project(SilentVibrance VERSION 2.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00)

set(CORE src/model.cpp src/config.cpp)
add_executable(SilentVibrance WIN32
  src/main.cpp src/ui.cpp src/process_monitor.cpp ${CORE}
  src/gpu/gpu_backend.cpp src/gpu/nvidia_backend.cpp src/gpu/amd_backend.cpp src/gpu/intel_backend.cpp
  resources/app.rc)
target_include_directories(SilentVibrance PRIVATE src src/gpu third_party/amd_adl resources)
target_link_libraries(SilentVibrance PRIVATE comctl32 comdlg32 shell32 ole32 advapi32 user32 gdi32 gdiplus dwmapi uxtheme)
if(MSVC)
  target_compile_options(SilentVibrance PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_options(SilentVibrance PRIVATE /OPT:REF /OPT:ICF)
else()
  target_compile_options(SilentVibrance PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(SilentVibrance PRIVATE -static "-Wl,--subsystem,windows")
endif()

enable_testing()
add_executable(SilentVibrance_tests tests/tests.cpp ${CORE})
target_include_directories(SilentVibrance_tests PRIVATE src)
target_link_libraries(SilentVibrance_tests PRIVATE shell32 ole32 advapi32)
if(NOT MSVC)
  target_link_options(SilentVibrance_tests PRIVATE -municode)
endif()
add_test(NAME core COMMAND SilentVibrance_tests)
add_executable(SilentVibrance_lifecycle tests/lifecycle.cpp)
target_link_libraries(SilentVibrance_lifecycle PRIVATE user32 advapi32 psapi)
if(NOT MSVC)
  target_link_options(SilentVibrance_lifecycle PRIVATE -municode)
endif()
add_test(NAME lifecycle COMMAND SilentVibrance_lifecycle $<TARGET_FILE:SilentVibrance>)
